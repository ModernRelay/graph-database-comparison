## 2. Two-hop query: projects Acme's employees contribute to

LOAD 'age';
SET search_path = ag_catalog, "$user", public;
SELECT * FROM cypher('cmp', $$
  MATCH (c:Company {slug: 'acme'})<-[:WorksAt]-(p:Person)-[:ContributesTo]->(j:Project)
  RETURN DISTINCT j.name AS project
  ORDER BY project
$$) AS (project agtype);

$ docker exec -i cmp-apache-age psql -X -q -U postgres -v ON_ERROR_STOP=1 -f -
